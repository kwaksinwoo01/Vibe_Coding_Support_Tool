function IsDigitChar(const C: WideChar): Boolean;
begin
  Result := False;
  if C < '0' then Exit;
  if C > '9' then Exit;
  Result := True;
end;


function IsDateSeparator(const C: WideChar): Boolean;
begin
  Result := (C = '_') or (C = ' ') or (C = '-') or (C = '.');
end;


function IsDateSeparatorAt(const S: WideString; Position: Integer): Boolean;
begin
  Result := False;
  if Position < 1 then Exit;
  if Position > Length(S) then Exit;
  Result := IsDateSeparator(S[Position]);
end;


function IsValidMonthDayAt(
  const S: WideString;
  MonthStart: Integer;
  DayStart: Integer
): Boolean;
var
  MonthTens, MonthOnes, DayTens, DayOnes: WideChar;
begin
  Result := False;


  if MonthStart < 1 then Exit;
  if DayStart < 1 then Exit;
  if MonthStart + 1 > Length(S) then Exit;
  if DayStart + 1 > Length(S) then Exit;


  MonthTens := S[MonthStart];
  MonthOnes := S[MonthStart + 1];
  DayTens := S[DayStart];
  DayOnes := S[DayStart + 1];


  if not IsDigitChar(MonthTens) then Exit;
  if not IsDigitChar(MonthOnes) then Exit;
  if not IsDigitChar(DayTens) then Exit;
  if not IsDigitChar(DayOnes) then Exit;


  if MonthTens = '0' then
  begin
    if MonthOnes < '1' then Exit;
    if MonthOnes > '9' then Exit;
  end
  else if MonthTens = '1' then
  begin
    if MonthOnes < '0' then Exit;
    if MonthOnes > '2' then Exit;
  end
  else
    Exit;


  if DayTens = '0' then
  begin
    if DayOnes < '1' then Exit;
    if DayOnes > '9' then Exit;
  end
  else if (DayTens = '1') or (DayTens = '2') then
  begin
    if DayOnes < '0' then Exit;
    if DayOnes > '9' then Exit;
  end
  else if DayTens = '3' then
  begin
    if DayOnes < '0' then Exit;
    if DayOnes > '1' then Exit;
  end
  else
    Exit;


  Result := True;
end;


function IsValidYYMMDDAt(const S: WideString; StartPos: Integer): Boolean;
var
  j: Integer;
begin
  Result := False;


  if StartPos < 1 then Exit;
  if StartPos + 5 > Length(S) then Exit;


  for j := StartPos to StartPos + 5 do
  begin
    if not IsDigitChar(S[j]) then Exit;
  end;


  if StartPos > 1 then
  begin
    if not IsDateSeparatorAt(S, StartPos - 1) then Exit;
  end;


  if StartPos + 5 < Length(S) then
  begin
    if not IsDateSeparatorAt(S, StartPos + 6) then Exit;
  end;


  if not IsValidMonthDayAt(S, StartPos + 2, StartPos + 4) then Exit;
  Result := True;
end;


function IsValidYYYYSeparatedDateAt(
  const S: WideString;
  StartPos: Integer;
  var CompactDate: WideString;
  var SourceToken: WideString
): Boolean;
var
  j: Integer;
  SeparatorChar: WideChar;
begin
  Result := False;
  CompactDate := '';
  SourceToken := '';


  if StartPos < 1 then Exit;
  if StartPos + 9 > Length(S) then Exit;


  for j := StartPos to StartPos + 3 do
  begin
    if not IsDigitChar(S[j]) then Exit;
  end;


  if not IsDigitChar(S[StartPos + 5]) then Exit;
  if not IsDigitChar(S[StartPos + 6]) then Exit;
  if not IsDigitChar(S[StartPos + 8]) then Exit;
  if not IsDigitChar(S[StartPos + 9]) then Exit;


  SeparatorChar := S[StartPos + 4];
  if SeparatorChar <> '-' then
  begin
    if SeparatorChar <> '.' then
    begin
      if SeparatorChar <> '_' then Exit;
    end;
  end;


  if S[StartPos + 7] <> SeparatorChar then Exit;


  if StartPos > 1 then
  begin
    if not IsDateSeparatorAt(S, StartPos - 1) then Exit;
  end;


  if StartPos + 9 < Length(S) then
  begin
    if not IsDateSeparatorAt(S, StartPos + 10) then Exit;
  end;


  if not IsValidMonthDayAt(S, StartPos + 5, StartPos + 8) then Exit;


  SourceToken := WideCopy(S, StartPos, 10);
  CompactDate :=
    WideCopy(S, StartPos + 2, 2) +
    WideCopy(S, StartPos + 5, 2) +
    WideCopy(S, StartPos + 8, 2);


  Result := True;
end;


function ExtractExistingDateToYYMMDD(
  const S: WideString;
  var SourceToken: WideString
): WideString;
var
  Position: Integer;
  CompactDate: WideString;
begin
  Result := '';
  SourceToken := '';


  if Length(S) >= 10 then
  begin
    for Position := 1 to Length(S) - 9 do
    begin
      CompactDate := '';
      if IsValidYYYYSeparatedDateAt(
        S,
        Position,
        CompactDate,
        SourceToken
      ) then
      begin
        Result := CompactDate;
        Exit;
      end;
    end;
  end;


  if Length(S) >= 6 then
  begin
    for Position := 1 to Length(S) - 5 do
    begin
      if IsValidYYMMDDAt(S, Position) then
      begin
        SourceToken := WideCopy(S, Position, 6);
        Result := SourceToken;
        Exit;
      end;
    end;
  end;
end;


function IsAsciiText(const S: WideString): Boolean;
var
  Position: Integer;
begin
  Result := True;
  for Position := 1 to Length(S) do
  begin
    if S[Position] > '~' then
    begin
      Result := False;
      Exit;
    end;
  end;
end;


procedure EnsureLogFile;
begin

  WideForceDirectories(TOOL_ROOT);


  if WideFileExists(LOG_PATH) then
  begin
    if WideFileSize(LOG_PATH) > MAX_LOG_BYTES then
    begin
      FileWriteContent(
        LOG_PATH,

      );

        UTF8Encode('[LOG ROTATED]' + #13#10)
      );
    end;
  end
  else
  begin
    FileWriteContent(

    );
  end;
end;

function PrepareConsoleDocumentPath(
  const SourcePath: WideString;
  var TempCopyCreated: Boolean
): WideString;
var
  ShortPath, TempPath, Extension: WideString;
begin
  Result := '';
  TempCopyCreated := False;


  ShortPath := WideExtractShortPathName(SourcePath);
  if Length(ShortPath) > 0 then
  begin
    if IsAsciiText(ShortPath) then
    begin
      Result := ShortPath;
      Exit;
    end;
  end;


  if not WideForceDirectories(TEMP_WORK_DIR) then Exit;


  Extension := WideLowerCase(WideExtractFileExt(SourcePath));
  TempPath :=
    TEMP_WORK_DIR +
    'input_' + IntToStr(GetCurrentFileIndex) + Extension;


  if WideCopyFile(SourcePath, TempPath, False) then
  begin
    Result := TempPath;
    TempCopyCreated := True;
  end;
end;


function ExtractDocumentText(
  const SourcePath: WideString;
  var ExtractorMeta: WideString;
  var ExtractorExitCode: Cardinal
): WideString;
var
  ConsoleInputPath, MetaPath: WideString;
  CommandLine, ConsoleOutput: String;
  TempCopyCreated: Boolean;
begin
  Result := '';
  ExtractorMeta := '';
  ExtractorExitCode := 127;
  ConsoleOutput := '';
  TempCopyCreated := False;


  if not WideFileExists(EXTRACTOR_SCRIPT) then
  begin
    ExtractorMeta := 'status=extractor_missing' + #13#10 +
      'extractor=' + EXTRACTOR_SCRIPT + #13#10;
    Exit;
  end;


  ConsoleInputPath :=
    PrepareConsoleDocumentPath(SourcePath, TempCopyCreated);


  if Length(ConsoleInputPath) = 0 then
  begin
    ExtractorMeta := 'status=input_path_prepare_failed' + #13#10;
    Exit;
  end;


  if not WideForceDirectories(TEMP_WORK_DIR) then
  begin
    ExtractorMeta := 'status=temp_directory_failed' + #13#10;
    Exit;
  end;


  MetaPath :=
    TEMP_WORK_DIR +
    'meta_' + IntToStr(GetCurrentFileIndex) + '.txt';


  if WideFileExists(MetaPath) then
    WideDeleteFile(MetaPath);


  CommandLine := UTF8Encode(
    PYTHON_COMMAND +
    ' "' + EXTRACTOR_SCRIPT + '"' +
    ' --input "' + ConsoleInputPath + '"' +
    ' --meta "' + MetaPath + '"' +
    ' --work-dir "' + TEMP_WORK_DIR + '"' +
    ' --max-pages 2' +
    ' --max-sheets 3' +
    ' --max-rows 200' +
    ' --max-columns 50' +
    ' --max-characters 50000' +
    ' --ocr-dpi 220' +
    ' --timeout 120'
  );


  ExtractorExitCode := ExecConsoleApp(CommandLine, ConsoleOutput);


  if Length(ConsoleOutput) > 0 then
    Result := UTF8Decode(ConsoleOutput);


  if WideFileExists(MetaPath) then
    ExtractorMeta := FileReadText(MetaPath)
  else
    ExtractorMeta := 'status=metadata_missing' + #13#10;


  if TempCopyCreated then
    WideDeleteFile(ConsoleInputPath);


  if WideFileExists(MetaPath) then
    WideDeleteFile(MetaPath);
end;


function NormalizeDocumentText(const S: WideString): WideString;
begin
  Result := WideLowerCase(S);
  Result := WideReplaceStr(Result, ' ', '');
  Result := WideReplaceStr(Result, #9, '');
  Result := WideReplaceStr(Result, #10, '');
  Result := WideReplaceStr(Result, #11, '');
  Result := WideReplaceStr(Result, #12, '');
  Result := WideReplaceStr(Result, #13, '');
end;


function ContainsKeywordWithGaps(
  const Text: WideString;
  const Keyword: WideString;
  MaxGap: Integer
): Boolean;
var
  StartPos, TextPos, KeywordPos, GapCount: Integer;
  MatchFailed: Boolean;
begin
  Result := False;


  if Length(Text) = 0 then Exit;
  if Length(Keyword) = 0 then Exit;


  for StartPos := 1 to Length(Text) do
  begin
    if Text[StartPos] = Keyword[1] then
    begin
      TextPos := StartPos + 1;
      KeywordPos := 2;
      MatchFailed := False;


      while KeywordPos <= Length(Keyword) do
      begin
        GapCount := 0;


        while TextPos <= Length(Text) do
        begin
          if Text[TextPos] = Keyword[KeywordPos] then Break;


          Inc(GapCount);
          if GapCount > MaxGap then
          begin
            MatchFailed := True;
            Break;
          end;


          Inc(TextPos);
        end;


        if MatchFailed then Break;


        if TextPos > Length(Text) then
        begin
          MatchFailed := True;
          Break;
        end;


        Inc(TextPos);
        Inc(KeywordPos);
      end;


      if not MatchFailed then
      begin
        Result := True;
        Exit;
      end;
    end;
  end;
end;


function SetEarlierPosition(
  CurrentPosition: Integer;
  CandidatePosition: Integer
): Integer;
begin
  Result := CurrentPosition;
  if CandidatePosition <= 0 then Exit;


  if Result = 0 then
    Result := CandidatePosition
  else if CandidatePosition < Result then
    Result := CandidatePosition;
end;

function ClassifyDocumentText(
  const DocumentText: WideString;
  var QuotationScore: Integer;
  var TransactionScore: Integer;
  var QuotationTitlePos: Integer;
  var TransactionTitlePos: Integer
): WideString;
var
  NormalizedText, TitleText: WideString;
begin
  Result := 'NA';
  QuotationScore := 0;
  TransactionScore := 0;
  QuotationTitlePos := 0;
  TransactionTitlePos := 0;


  NormalizedText := NormalizeDocumentText(DocumentText);


  if Length(NormalizedText) > 12000 then
    NormalizedText := WideCopy(NormalizedText, 1, 12000);


  TitleText := NormalizedText;
  if Length(TitleText) > 2500 then
    TitleText := WideCopy(TitleText, 1, 2500);


  QuotationTitlePos := FindFirstQuotationTitle(TitleText);
  TransactionTitlePos := FindFirstTransactionTitle(TitleText);


  if QuotationTitlePos > 0 then
  begin
    if TransactionTitlePos = 0 then
    begin
      Result := '00.견적서';
      Exit;
    end;
  end;


  if TransactionTitlePos > 0 then
  begin
    if QuotationTitlePos = 0 then
    begin
      Result := '01.거래명세서';
      Exit;
    end;
  end;


  if QuotationTitlePos > 0 then
  begin
    if TransactionTitlePos > 0 then
    begin
      if QuotationTitlePos < TransactionTitlePos then
        Result := '00.견적서'
      else if TransactionTitlePos < QuotationTitlePos then
        Result := '01.거래명세서';


      if Result <> 'NA' then Exit;
    end;
  end;


  QuotationScore := GetQuotationSupportScore(NormalizedText);
  TransactionScore := GetTransactionSupportScore(NormalizedText);


  if QuotationScore < MIN_SUPPORT_SCORE then
  begin
    if TransactionScore < MIN_SUPPORT_SCORE then Exit;
  end;


  if QuotationScore > TransactionScore then
  begin
    Result := '00.견적서';
    Exit;
  end;


  if TransactionScore > QuotationScore then
  begin
    Result := '01.거래명세서';
    Exit;
  end;
end;

function AnalyzeSupportedDocument(
  const AFileName: WideString;
  const AFilePath: WideString;
  var ExtractorMeta: WideString;
  var ExtractorExitCode: Cardinal;
  var ExtractedTextLength: Integer;
  var QuotationScore: Integer;
  var TransactionScore: Integer;
  var QuotationTitlePos: Integer;
  var TransactionTitlePos: Integer;
  var DetectedBy: WideString
): WideString;
var
  DocumentText: WideString;
begin
  Result := 'NA';
  ExtractedTextLength := 0;
  QuotationScore := 0;
  TransactionScore := 0;
  QuotationTitlePos := 0;
  TransactionTitlePos := 0;
  DetectedBy := 'none';


  DocumentText := ExtractDocumentText(
    AFilePath,
    ExtractorMeta,
    ExtractorExitCode
  );


  ExtractedTextLength := Length(DocumentText);


  if ExtractedTextLength > 0 then
  begin
    Result := ClassifyDocumentText(
      DocumentText,
      QuotationScore,
      TransactionScore,
      QuotationTitlePos,
      TransactionTitlePos
    );


    if Result <> 'NA' then
    begin
      DetectedBy := 'document_content';
      Exit;
    end;
  end;


  Result := InferDocumentTypeFromFileName(AFileName);
  if Result <> 'NA' then
    DetectedBy := 'filename_fallback';
end;

function CustomWideTrim(const S: WideString): WideString;
var
  First, Last: Integer;
begin
  First := 1;
  while First <= Length(S) do
  begin
    if S[First] <> ' ' then Break;
    Inc(First);
  end;


  Last := Length(S);
  while Last >= First do
  begin
    if S[Last] <> ' ' then Break;
    Dec(Last);
  end;


  if Last < First then
    Result := ''
  else
    Result := WideCopy(S, First, Last - First + 1);
end;


function RemoveDocumentLabels(const S: WideString): WideString;
begin

function NormalizeRemainder(const S: WideString): WideString;
begin
  Result := S;


  while WidePos('  ', Result) > 0 do
    Result := WideReplaceStr(Result, '  ', ' ');


  while WidePos('__', Result) > 0 do
    Result := WideReplaceStr(Result, '__', '_');


  while WidePos('_ ', Result) > 0 do
    Result := WideReplaceStr(Result, '_ ', '_');


  while WidePos(' _', Result) > 0 do
    Result := WideReplaceStr(Result, ' _', '_');


  Result := CustomWideTrim(Result);


  while Length(Result) > 0 do
  begin
    if Result[1] <> '_' then Break;
    Result := WideCopy(Result, 2, Length(Result) - 1);
  end;


  while Length(Result) > 0 do
  begin
    if Result[Length(Result)] <> '_' then Break;
    Result := WideCopy(Result, 1, Length(Result) - 1);
  end;


  Result := CustomWideTrim(Result);
end;
